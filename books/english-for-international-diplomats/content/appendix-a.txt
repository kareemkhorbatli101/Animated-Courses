Sub PlayDialogue()
    ' Plays dialogue lines with gender-appropriate voices
    ' Male voices: blue text (color: 255)
    ' Female voices: purple text (color: 10040115)
    
    Dim oPara As Paragraph
    Dim oRun As Range
    Dim voice As Object
    Set voice = CreateObject("SAPI.SpVoice")
    
    ' Get voices
    Dim voices As Object
    Set voices = voice.GetVoices
    
    ' Find male and female voices
    Dim maleVoice As Object
    Dim femaleVoice As Object
    
    For i = 0 To voices.Count - 1
        If InStr(voices.Item(i).GetDescription, "Female") > 0 Or _
           InStr(voices.Item(i).GetDescription, "Zira") > 0 Then
            Set femaleVoice = voices.Item(i)
        ElseIf InStr(voices.Item(i).GetDescription, "Male") > 0 Or _
               InStr(voices.Item(i).GetDescription, "David") > 0 Then
            Set maleVoice = voices.Item(i)
        End If
    Next i
    
    ' Set default to first voice if not found
    If maleVoice Is Nothing Then Set maleVoice = voices.Item(0)
    If femaleVoice Is Nothing Then Set femaleVoice = voices.Item(0)
    
    ' Play each dialogue line based on text color
    For Each oPara In Selection.Paragraphs
        For Each oRun In oPara.Range.Sentences
            If oRun.Font.Color = 10040115 Then ' Purple = Female
                voice.Voice = femaleVoice
            Else
                voice.Voice = maleVoice ' Blue or default = Male
            End If
            
            voice.Speak oRun.Text, 0
        Next oRun
    Next oPara
End Sub
