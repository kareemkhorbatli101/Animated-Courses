Sub PlayDialogue()
    ' Plays dialogue lines with gender-appropriate voices
    ' Male voices: blue text (color: 255)
    ' Female voices: purple text (color: 10040115)
    
    Dim oPara As Paragraph
    Dim oRun As Range
    Dim voice As Object
    Set voice = CreateObject("SAPI.SpVoice")
    
    ' Get voices
    Dim voices As Object
    Set voices = voice.GetVoices
    
    ' Find male and female voices
    Dim maleVoice As Object
    Dim femaleVoice As Object
    
    For i = 0 To voices.Count - 1
        If InStr(voices.Item(i).GetDescription, "Female") > 0 Or _
           InStr(voices.Item(i).GetDescription, "Zira") > 0 Then
            Set femaleVoice = voices.Item(i)
        ElseIf InStr(voices.Item(i).GetDescription, "Male") > 0 Or _
               InStr(voices.Item(i).GetDescription, "David") > 0 Then
            Set maleVoice = voices.Item(i)
        End If
    Next i
    
    ' Set default to first voice if not found
    If maleVoice Is Nothing Then Set maleVoice = voices.Item(0)
    If femaleVoice Is Nothing Then Set femaleVoice = voices.Item(0)
    
    ' Play each dialogue line based on text color
    For Each oPara In Selection.Paragraphs
        For Each oRun In oPara.Range.Sentences
            If oRun.Font.Color = 10040115 Then ' Purple = Female
                voice.Voice = femaleVoice
            Else
                voice.Voice = maleVoice ' Blue or default = Male
            End If
            
            voice.Speak oRun.Text, 0
        Next oRun
    Next oPara
End Sub

@TRACK 🔊 Track 4.2 - Exercise 2: Group Brainstorm – Types of Written Communication at the Ministry
@SCRIPT
~Senior Officer: The thing to understand about a note verbale is that it is not a letter. A note verbale is written in the third person throughout — no "I" or "we." Instead: "The Ministry of Foreign Affairs of the Republic of Kazakhstan presents its compliments to the Embassy of the Federal Republic of Germany and has the honor to inform..." Every note verbale begins with this formula. Every note verbale ends with a closing compliments formula: "The Ministry of Foreign Affairs avails itself of this opportunity to renew to the Embassy the assurances of its highest consideration." These are not optional. They are the form through which the institution speaks — and getting them right signals that the diplomatic service knows what it is doing.
^Junior Officer: And if I write it in first person?
~Senior Officer: Then it is not a note verbale. It is a letter — and a letter carries different weight and a different institutional meaning. The distinction matters to the receiving party. Always use the correct form for the correct purpose.
~Senior Officer: Once the note verbale is cleared, it goes to Consular Affairs for authentication before dispatch. Nurlan's section handles that — have your document ready 24 hours before it needs to leave the building.
